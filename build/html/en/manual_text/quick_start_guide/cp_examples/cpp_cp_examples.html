C++ CP API¶

This section presents a simple C++ example to demonstrate how to write a program using the OptVerse CP Solver’s C++ API.

This example illustrates how to:

  • Set solver parameters

  • Construct a model

  • Write the constructed problem to file

  • Optimize the model

  • Print solution information

The mathematical formulation for the problem to be solved is as follows:

\[\begin{split}\text{minimize } & b_1 + b_2 + b_3 + b_4 + 2x_1 + 2x_2 \\ \text{subject to } & -x_1 + 4x_2 + 8 \leq 15 \\ & 3b_1 - b_2 + x_1 = [3, 4] \cup [6, 10] \cup [12, 14] \\ & b_1 \lor b_2 \lor b_4 \\ & \neg b_1 \land b_3 \\ & b_2 + b_3 \geq 1 \\ & b_2 + b_4 \leq 1 \\ & b_2 + b_4 = 1 \\ & b_1 \rightarrow b_4 \\ \text{where } & b_1, b_2, b_3, b_4 \in [0, 1] \\ & x_1 \in [2, 5] \\ & x_2 \in [1, 2] \cup [5, 7] \cup [15, 15] \\\end{split}\]
#include "optv_c++.h"

        OPTVEnv env;

        env.Set(OPTVDblParam::CP_TIME_LIMIT, 7200);

        OPTVCPModel model(env);

        model.Set(OPTVIntParam::CP_THREADS, 12);

        OPTVCPVar b1 = model.NewBoolVar("b1");
        OPTVCPVar b2 = model.NewBoolVar("b2");
        OPTVCPVar b3 = model.NewBoolVar("b3");
        OPTVCPVar b4 = model.NewBoolVar("b4");
        OPTVCPVar x1 = model.NewIntVar(2, 5, "x1");
        // domain: [1, 2], [5, 7], [15, 15]
        OPTVCPVar x2 = model.NewIntVarFromDomain({1, 2, 5, 7, 15, 15}, "x2");
        OPTVCPVar c1 = model.NewConstant(11);

        OPTVCPConstr lin0 = model.AddLinearConstraint(-x1 + 4 * x2 + c1 - 3, -OPTV_CP_INF, 15);
        // domain: [3, 4], [6, 10], [12, 14]
        std::vector<int64_t> domain = {3, 4, 6, 10, 12, 14};
        OPTVCPConstr lin1 = model.AddLinearConstraintInDomain(3 * b1 - b2 + x1, domain);

        OPTVCPConstr boolOr = model.AddBoolOr({b1, b2, b4});
        OPTVCPConstr boolAnd = model.AddBoolAnd({~b1, b3});
        OPTVCPConstr boolALO = model.AddAtLeastOne({b2, b3});
        OPTVCPConstr boolAMO = model.AddAtMostOne({b2, b4});
        OPTVCPConstr boolEO = model.AddExactlyOne({b2, b4});
        OPTVCPConstr boolImpl = model.AddImplication({b1, b4});

        boolOr.OnlyEnforceIf({b3, b4});

        model.Write("cp.fzn");

        OPTVCPSolver solver;
        auto status = solver.Solve(model);

        switch (status) {
            case OPTV_CP_UNKNOWN:
                cout << "Model is unknown." << endl;
                break;
            case OPTV_CP_MODEL_INVALID:
                cout << "Model is invalid." << endl;
                break;
            case OPTV_CP_FEASIBLE:
                cout << "Model is feasible." << endl;
                break;
            case OPTV_CP_INFEASIBLE:
                cout << "Model is infeasible." << endl;
                break;
            case OPTV_CP_OPTIMAL: {
                std::vector<OPTVCPVar> vars{b1, b2, b3, b4, x1, x2};
                const auto& result = solver.GetValues(vars);
                cout << "Optimal solution found. Best solution: " << endl;
                for (int i = 0; i < result.size(); ++i) {
                    cout << "  " << vars[i].GetName() << " = " << result[i] << endl;
                }

                double objVal = solver.GetObjectiveValue();
                break;
            }
            case OPTV_CP_USER_INTERRUPT:
                cout << "Solver was interrupted by the user." << endl;
                break;
        }

/*
 * Copyright (c) Huawei Technologies Co., Ltd. 2025. All rights reserved.
 * Description: OptVerse c++ interface example - build and solve a CP-SAT problem
 */
#include "optv_c++.h"
using namespace std;

int main()
{
    try {
        OPTVEnv env;
        // Before model construction: set parameters through environment
        env.Set(OPTVDblParam::CP_TIME_LIMIT, 7200);

        OPTVCPModel model(env);
        // After model construction: set parameters through model
        model.Set(OPTVIntParam::CP_THREADS, 12);

        // Add variables to the model
        OPTVCPVar b1 = model.NewBoolVar("b1");
        OPTVCPVar b2 = model.NewBoolVar("b2");
        OPTVCPVar b3 = model.NewBoolVar("b3");
        OPTVCPVar b4 = model.NewBoolVar("b4");
        OPTVCPVar x1 = model.NewIntVar(2, 5, "x1");
        // domain: [1, 2], [5, 7], [15, 15]
        OPTVCPVar x2 = model.NewIntVarFromDomain({1, 2, 5, 7, 15, 15}, "x2");
        OPTVCPVar c1 = model.NewConstant(11);

        // Add constraint to the model
        OPTVCPConstr lin0 = model.AddLinearConstraint(-x1 + 4 * x2 + c1 - 3, -OPTV_CP_INF, 15);
        // domain: [3, 4], [6, 10], [12, 14]
        std::vector<int64_t> domain = {3, 4, 6, 10, 12, 14};
        OPTVCPConstr lin1 = model.AddLinearConstraintInDomain(3 * b1 - b2 + x1, domain);

        OPTVCPConstr boolOr = model.AddBoolOr({b1, b2, b4});
        OPTVCPConstr boolAnd = model.AddBoolAnd({~b1, b3});
        OPTVCPConstr boolALO = model.AddAtLeastOne({b2, b3});
        OPTVCPConstr boolAMO = model.AddAtMostOne({b2, b4});
        OPTVCPConstr boolEO = model.AddExactlyOne({b2, b4});
        OPTVCPConstr boolImpl = model.AddImplication({b1, b4});

        boolOr.OnlyEnforceIf({b3, b4});

        model.Write("cp.fzn");
        OPTVCPSolver solver;
        auto status = solver.Solve(model);

        switch (status) {
            case OPTV_CP_UNKNOWN:
                cout << "Model is unknown." << endl;
                break;
            case OPTV_CP_MODEL_INVALID:
                cout << "Model is invalid." << endl;
                break;
            case OPTV_CP_FEASIBLE:
                cout << "Model is feasible." << endl;
                break;
            case OPTV_CP_INFEASIBLE:
                cout << "Model is infeasible." << endl;
                break;
            case OPTV_CP_OPTIMAL: {
                std::vector<OPTVCPVar> vars{b1, b2, b3, b4, x1, x2};
                const auto& result = solver.GetValues(vars);
                cout << "Optimal solution found. Best solution: " << endl;
                for (int i = 0; i < result.size(); ++i) {
                    cout << "  " << vars[i].GetName() << " = " << result[i] << endl;
                }

                double objVal = solver.GetObjectiveValue();
                break;
            }
            case OPTV_CP_USER_INTERRUPT:
                cout << "Solver was interrupted by the user." << endl;
                break;
        }

    } catch (const OPTVException& e) {
        cout << e.GetErrorCode() << ": " << e.GetMessage() << endl;
    } catch (...) {
        cerr << "Unknown exception" << endl;
    }
    return 0;
}

Include the Header¶

To provide access to the C++ interface, include the header file optv_c++.h.

#include "optv_c++.h"

Create the Environment¶

The first step is to create the environment object.

OPTVEnv env;

Configuring Parameters Before Model Construction¶

To edit solver parameters before calling the OPTVCPModel constructor, a user must set the parameter through the environment object. In this example, we demonstrate setting the solver time limit through the environment, prior to creation of the model object.

env.Set(OPTVDblParam::CP_TIME_LIMIT, 7200);

Configuring Parameters After Model Construction¶

To edit solver parameters after calling the OPTVCPModel constructor, a user must set the parameter through the model object. In this example, we demonstrate setting the number of threads through the model.

model.Set(OPTVIntParam::CP_THREADS, 12);

Creating the Model¶

After creating the environment, the next step is to create the model. Each OptVerse model is an optimization problem consisting of a set of variables, a set of constraints, and associated attributes (i.e., variable or constraint bounds, solution status, variable or constraint names, etc.).

Construct An Empty Model¶

If a user wishes to construct a model from scratch using API calls, they will first need to create an empty model.

OPTVCPModel model(env);

Loading Model from File¶

If the .mps [1], .lp [2], or .fzn [3] extension is passed as the second parameter, the problem file is read. This allows solving or modifying a pre-existing model through the API.

OPTVEnv env;
OPTVCPModel(env, fileName);
[1]

 For a description of the MPS format, see File Formats.

[2]

 For a description of the LP format, see File Formats.

[3]

 For a description of the FZN format, see File Formats.

Add Variables to Model¶

If an empty model was created in the previous step, variables and constraints must be added to the model before it can be solved. Alternatively, a model loaded from a file can be expanded through the addition of variables or constraints.

The following methods can be used to add variables to the model:

  • OPTVCPModel::NewIntVar()

  • OPTVCPModel::NewIntVarFromDomain()

  • OPTVCPModel::NewBoolVar()

  • OPTVCPModel::NewConstant()

OPTVCPVar b1 = model.NewBoolVar("b1");
OPTVCPVar b2 = model.NewBoolVar("b2");
OPTVCPVar b3 = model.NewBoolVar("b3");
OPTVCPVar b4 = model.NewBoolVar("b4");
OPTVCPVar x1 = model.NewIntVar(2, 5, "x1");
// domain: [1, 2], [5, 7], [15, 15]
OPTVCPVar x2 = model.NewIntVarFromDomain({1, 2, 5, 7, 15, 15}, "x2");
OPTVCPVar c1 = model.NewConstant(11);

Add Constraints to Model¶

After adding variables, a user may now add constraints to the model. The OptVerse CP API provides several types of constraints:

  • Linear;

  • Boolean Or;

  • Boolean And;

  • Boolean Xor;

  • Implication;

  • At Least One;

  • At Most One;

  • Exactly One;

  • Table;

  • Interval;

  • No Overlap;

  • Cumulative;

  • All Different.

OPTVCPConstr lin0 = model.AddLinearConstraint(-x1 + 4 * x2 + c1 - 3, -OPTV_CP_INF, 15);
// domain: [3, 4], [6, 10], [12, 14]
std::vector<int64_t> domain = {3, 4, 6, 10, 12, 14};
OPTVCPConstr lin1 = model.AddLinearConstraintInDomain(3 * b1 - b2 + x1, domain);

OPTVCPConstr boolOr = model.AddBoolOr({b1, b2, b4});
OPTVCPConstr boolAnd = model.AddBoolAnd({~b1, b3});
OPTVCPConstr boolALO = model.AddAtLeastOne({b2, b3});
OPTVCPConstr boolAMO = model.AddAtMostOne({b2, b4});
OPTVCPConstr boolEO = model.AddExactlyOne({b2, b4});
OPTVCPConstr boolImpl = model.AddImplication({b1, b4});

boolOr.OnlyEnforceIf({b3, b4});

Model Optimization¶

Once the model is constructed and any necessary parameters have been set, the model can be solved by invoking OPTVCPSolver::Solve(). After optimization, the solution status and optimal solution are saved in the OPTVCPSolver object.

OPTVCPSolver solver;
auto status = solver.Solve(model);

Querying the Solution¶

To output the solution after optimization, the OPTVCPSolver can be used to get data from the solved model. First, the function OPTVCPSolver::Solve() returns the status, which is used to check if a feasible solution has been found. We obtain the best found objective value by using the OPTVCPSolver::GetObjectiveValue() method. The value of a variable can be obtained with the OPTVCPSolver::GetSolution(), OPTVCPSolver::GetValue() or OPTVCPSolver::GetValues() methods.

switch (status) {
    case OPTV_CP_UNKNOWN:
        cout << "Model is unknown." << endl;
        break;
    case OPTV_CP_MODEL_INVALID:
        cout << "Model is invalid." << endl;
        break;
    case OPTV_CP_FEASIBLE:
        cout << "Model is feasible." << endl;
        break;
    case OPTV_CP_INFEASIBLE:
        cout << "Model is infeasible." << endl;
        break;
    case OPTV_CP_OPTIMAL: {
        std::vector<OPTVCPVar> vars{b1, b2, b3, b4, x1, x2};
        const auto& result = solver.GetValues(vars);
        cout << "Optimal solution found. Best solution: " << endl;
        for (int i = 0; i < result.size(); ++i) {
            cout << "  " << vars[i].GetName() << " = " << result[i] << endl;
        }

        double objVal = solver.GetObjectiveValue();
        break;
    }
    case OPTV_CP_USER_INTERRUPT:
        cout << "Solver was interrupted by the user." << endl;
        break;
}

Compiling and Running¶

Several example files for the constraint programming solver have been provided in ${OPTV_HOME}/examples/c++ along with the CMake file for compiling the examples on Linux.

To compile and run the examples, go to the ${OPTV_HOME}/examples directory and execute the following:

mkdir build && cd build
cmake ..
make -j $(nproc)

After compilation, the example from this section can be run by calling:

./cp
